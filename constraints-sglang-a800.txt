# 安装顺序（GPU 栈强烈建议按序装）：
#   1) uv python install 3.12 && uv venv --python 3.12
#   2) torch 栈：uv pip install torch==2.10.0 torchaudio==2.10.0 \
#        (默认 PyPI wheel 已带 CUDA runtime；若要显式 cu128：
#         --index-url https://download.pytorch.org/whl/cu128)
#   3) GPU kernel：uv pip install flashinfer-python==0.6.7.post2 \
#        flashinfer-cubin==0.6.7.post2 sglang-kernel==0.4.1
#   4) sglang 本体（--no-deps 防止它把 torch 拽回 2.9.1）：
#        uv pip install --no-deps sglang==0.5.10
#   5) 其余 sglang 依赖 + 项目依赖，全部走本 constraints：
#        uv pip install -c constraints-sglang-a800.txt \
#          xgrammar outlines cuda-python torchao torch_memory_saver \
#          nvidia-cutlass-dsl quack-kernels timm sentencepiece \
#          faiss-gpu-cu12 pyserini spacy numpy pyarrow
#   6) flash-attn-4（如需）：装预编译 wheel，或确保 nvcc 在 PATH
#
# 注意事项：
#   - nvcc 不在 PATH（只有 /usr/local/cuda-12.2）。需要现场编译扩展时：

#       export LD_LIBRARY_PATH=/usr/local/cuda-12.2/lib64:$LD_LIBRARY_PATH
#   - sglang 0.5.10 官方钉 torch 2.9.1 / transformers 5.3.0；你保留 2.10.0 / 5.5.4
